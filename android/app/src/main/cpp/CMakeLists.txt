cmake_minimum_required(VERSION 3.22)
project(bookstr_torrent LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
# Same immutable libtorrent revision as scripts/build-ios-torrent.sh.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(deprecated-functions OFF CACHE BOOL "" FORCE)
set(logging OFF CACHE BOOL "" FORCE)
set(i2p OFF CACHE BOOL "" FORCE)
set(mutable-torrents OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenSSL TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_GnuTLS TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_LibGcrypt TRUE)
set(Boost_NO_BOOST_CMAKE ON)
if(DEFINED ENV{BOOST_ROOT})
    set(Boost_INCLUDE_DIR "$ENV{BOOST_ROOT}/include" CACHE PATH "Boost headers")
endif()
set_property(GLOBAL PROPERTY JOB_POOLS compile=2)
set(CMAKE_JOB_POOL_COMPILE compile)
add_compile_definitions(TORRENT_HAVE_MMAP=0)
FetchContent_Declare(libtorrent
    GIT_REPOSITORY https://github.com/arvidn/libtorrent.git
    GIT_TAG 9d7443f467147d1784fb7516d2a882db1abb5a8b
    GIT_SUBMODULES deps/try_signal)
FetchContent_MakeAvailable(libtorrent)
add_library(bookstr_torrent SHARED torrent_jni.cpp)
target_link_libraries(bookstr_torrent PRIVATE torrent-rasterbar log)
target_link_options(bookstr_torrent PRIVATE "-Wl,-z,max-page-size=16384")
